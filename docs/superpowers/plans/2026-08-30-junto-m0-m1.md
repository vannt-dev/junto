# junto M0 + M1 — Kế hoạch triển khai

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Dựng task engine của junto chạy được thật — state bền trên đĩa, quality gate dựa trên bằng chứng, plugin cài được — **không có model ngoài**.

**Architecture:** pnpm workspace. `packages/core` chứa toàn bộ logic thuần (schema, chính sách chuyển phase, chạy gate). `packages/mcp` bọc core thành MCP server stdio với 3 tool. `src-hooks/` là 4 hook, bundle bằng esbuild thành file JS đơn lẻ trong `plugin/`. Plugin tự chứa hoàn toàn: không npx, không binary, không tải gì.

**Tech Stack:** TypeScript, Node >= 20, pnpm workspace, zod (schema), execa (spawn), esbuild (bundle), vitest (test), `@modelcontextprotocol/sdk` (MCP server).

**Spec:** `docs/superpowers/specs/2026-08-30-junto-design.md`

## Global Constraints

Mọi task đều ngầm mang các ràng buộc sau. Chép nguyên văn từ spec.

- **R1 — Chỉ ghi trong project.** Không code nào được tạo/sửa file ngoài `<project>/.junto/`. Không đụng `~/.claude`, `settings.json`, hay `.gitignore` gốc.
- **R2 — Không tải mã về chạy.** Không binary, không `chmod`, không CDN, không `npx` lúc khởi động phiên.
- **R3 — Không lưu bí mật.** Không ghi giá trị API key vào bất kỳ file nào.
- **R5 — Gate tất định, panel tư vấn.** Chỉ mã thoát của lệnh mới chặn được chuyển phase.
- **Node >= 20**, TypeScript strict mode bật.
- **`SCHEMA_VERSION = 1`.** File có `schemaVersion` lớn hơn → tool từ chối kèm thông báo, không đoán.
- **`canEnter()` phải thuần.** Mọi dữ kiện filesystem truyền vào qua tham số, không đọc đĩa bên trong.
- **`.gitattributes` phải có `* text=auto eol=lf`** ngay từ Task 2 — CI chạy ma trận Windows + Linux và có job `git diff --exit-code` trên artifact build; thiếu dòng này job đó đỏ vĩnh viễn.
- **Spawn luôn dùng argv dạng mảng**, không bao giờ ghép chuỗi shell.
- Commit message tiếng Việt, dạng conventional commit.

`packages/core/src/schema.ts`:

```ts
import { z } from "zod"

export const SCHEMA_VERSION = 1

export const DEFAULT_STALE_IGNORE = ["**/*.md", "docs/**", ".junto/**"]

export class SchemaVersionError extends Error {
  constructor(public found: number, public supported: number) {
    super(
      `File junto có schemaVersion ${found} nhưng bản junto này chỉ hiểu tới ${supported}. `
      + `Hãy nâng junto lên bản mới hơn. junto không đoán định dạng nó chưa biết.`,
    )
    this.name = "SchemaVersionError"
  }
}

function assertVersion(raw: unknown): void {
  const v = (raw as { schemaVersion?: unknown })?.schemaVersion
  if (typeof v === "number" && v > SCHEMA_VERSION) throw new SchemaVersionError(v, SCHEMA_VERSION)
}

export const phaseSchema = z.enum(["brief", "plan", "build", "verify", "done"])
export const sizeSchema = z.enum(["small", "standard", "deep"])
export const gateStateSchema = z.enum(["pass", "fail", "skipped"])

export const phaseStatusSchema = z.object({
  status: z.enum(["pending", "active", "done"]),
  at: z.string().optional(),
  approvedBy: z.literal("user").optional(),
  step: z.number().int().optional(),
  of: z.number().int().optional(),
})

export const gateStatusSchema = z.object({
  required: z.boolean(),
  verdict: z.string().nullable(),
  stale: z.boolean(),
  failStreak: z.number().int().min(0),
})

export const decisionSchema = z.object({ at: z.string(), what: z.string(), why: z.string() })

export const taskSchema = z.object({
  schemaVersion: z.number().int(),
  id: z.string().min(1),
  title: z.string(),
  size: sizeSchema,
  phase: phaseSchema,
  baseCommit: z.string().nullable(),
  createdAt: z.string(),
  updatedAt: z.string(),
  phases: z.record(z.string(), phaseStatusSchema),
  gates: z.record(z.string(), gateStatusSchema),
  decisions: z.array(decisionSchema),
  consults: z.array(z.string()),
})

export const gateSpecSchema = z.object({
  argv: z.array(z.string()).min(1),
  required: z.boolean(),
  timeoutMs: z.number().int().positive().optional(),
})

export const configSchema = z.object({
  schemaVersion: z.number().int(),
  gates: z.record(z.string(), gateSpecSchema),
  staleIgnore: z.array(z.string()).default(DEFAULT_STALE_IGNORE),
  autoApprove: z.array(sizeSchema).default([]),
}).passthrough()

export type Phase = z.infer<typeof phaseSchema>
export type Size = z.infer<typeof sizeSchema>
export type GateState = z.infer<typeof gateStateSchema>
export type PhaseStatus = z.infer<typeof phaseStatusSchema>
export type GateStatus = z.infer<typeof gateStatusSchema>
export type Decision = z.infer<typeof decisionSchema>
export type Task = z.infer<typeof taskSchema>
export type GateSpec = z.infer<typeof gateSpecSchema>
export type Config = z.infer<typeof configSchema>

export interface VerdictFile {
  gate: string
  argv: string[]
  cwd: string
  exitCode: number | null
  state: GateState
  startedAt: string
  durationMs: number
  outputTail: string
  outputBytes: number
  outputFile: string
  runner: string
  reason?: string
}

export function parseTask(raw: unknown): Task {
  assertVersion(raw)
  return taskSchema.parse(raw)
}

export function parseConfig(raw: unknown): Config {
  assertVersion(raw)
  return configSchema.parse(raw)
}
```

```ts
import type { GateState, Phase, Size, Task } from "./schema.js"

export type TransitionCheck = { ok: true } | { ok: false; reason: string }

export interface TransitionContext {
  /** brief.md tồn tại và có nội dung sau khi trim */
  briefNonEmpty: boolean
  /** plan.md tồn tại */
  planExists: boolean
  /** config.autoApprove */
  autoApprove: Size[]
  /** tên gate -> state đọc từ file verdict; null nếu chưa có verdict */
  verdictStates: Record<string, GateState | null>
}

const SMALL_PHASES: Phase[] = ["build", "verify", "done"]
const FULL_PHASES: Phase[] = ["brief", "plan", "build", "verify", "done"]

/**
 * Phase bắt buộc theo cỡ.
 * M1: `deep` giống `standard`. Phase panel/review của `deep` thuộc M3.
 */
export function requiredPhases(size: Size): Phase[] {
  return size === "small" ? [...SMALL_PHASES] : [...FULL_PHASES]
}

export function canEnter(task: Task, to: Phase, ctx: TransitionContext): TransitionCheck {
  const order = requiredPhases(task.size)
  const from = order.indexOf(task.phase)
  const target = order.indexOf(to)

  if (target === -1) return { ok: false, reason: `Phase "${to}" không có trong vòng đời của cỡ "${task.size}".` }
  if (from === -1) return { ok: false, reason: `Task đang ở phase "${task.phase}", không thuộc vòng đời của cỡ "${task.size}".` }
  if (target !== from + 1) {
    return { ok: false, reason: `Chỉ chuyển được sang phase liền kề. Đang ở "${task.phase}", muốn sang "${to}".` }
  }

  // L1
  if (task.phase === "brief" && to === "plan") {
    if (!ctx.briefNonEmpty) return { ok: false, reason: "brief.md chưa tồn tại hoặc còn rỗng." }
    return { ok: true }
  }

  // L2 — điểm dừng cứng duy nhất
  if (task.phase === "plan" && to === "build") {
    if (!ctx.planExists) return { ok: false, reason: "plan.md chưa tồn tại." }
    if (ctx.autoApprove.includes(task.size)) return { ok: true }
    if (task.phases.plan?.approvedBy !== "user") {
      return { ok: false, reason: "Plan chưa được duyệt. Người dùng phải gõ /junto:approve — model không tự duyệt được." }
    }
    return { ok: true }
  }

  // L3
  if (task.phase === "build" && to === "verify") return { ok: true }

  // L4
  if (task.phase === "verify" && to === "done") {
    const blocked: string[] = []
    for (const [name, status] of Object.entries(task.gates)) {
      if (!status.required) continue
      if (status.verdict === null) { blocked.push(`${name} (chưa chạy)`); continue }
      if (status.stale) { blocked.push(`${name} (bằng chứng hết hạn — code đã đổi)`); continue }
      const state = ctx.verdictStates[name] ?? null
      if (state !== "pass") blocked.push(`${name} (${state ?? "không đọc được verdict"})`)
    }
    if (blocked.length > 0) {
      return { ok: false, reason: `Gate bắt buộc chưa đạt: ${blocked.join(", ")}. Chạy /junto:verify.` }
    }
    return { ok: true }
  }

  return { ok: false, reason: `Không có luật cho chuyển "${task.phase}" -> "${to}".` }
}
```

```ts
/** Chuẩn hoá về dấu `/`, bỏ tiền tố `./`. */
function normalize(p: string): string {
  return p.replace(/\\/g, "/").replace(/^\.\//, "")
}

/** Dịch glob sang regex. Hỗ trợ `**`, `*`, `?`. `*` không vượt qua dấu `/`. */
function globToRegExp(glob: string): RegExp {
  let out = "^"
  for (let i = 0; i < glob.length; i++) {
    const c = glob[i]
    if (c === "*") {
      if (glob[i + 1] === "*") {
        // `**/` nuốt luôn cả trường hợp không có thư mục nào
        if (glob[i + 2] === "/") { out += "(?:.*/)?"; i += 2 }
        else { out += ".*"; i += 1 }
      } else {
        out += "[^/]*"
      }
    } else if (c === "?") {
      out += "[^/]"
    } else {
      out += c!.replace(/[.+^${}()|[\]\\]/g, "\\$&")
    }
  }
  return new RegExp(`${out}$`)
}

/**
 * Sửa file này có làm hết hạn verdict không?
 * Trả false khi đường dẫn khớp bất kỳ mẫu nào trong staleIgnore.
 */
export function shouldStale(relPath: string, staleIgnore: string[]): boolean {
  const p = normalize(relPath)
  return !staleIgnore.some(pattern => globToRegExp(normalize(pattern)).test(p))
}
```

```ts
import { mkdirSync, writeFileSync } from "node:fs"
import { join } from "node:path"
import { execa } from "execa"
import { taskDir } from "./paths.js"
import type { GateSpec, GateState, VerdictFile } from "./schema.js"

export const OUTPUT_TAIL_BYTES = 8192
const DEFAULT_TIMEOUT_MS = 300_000

export interface RunGateOptions {
  root: string
  taskId: string
  name: string
  spec: GateSpec
  runner: string
}

/** Lấy phần đuôi, cắt theo byte nhưng không cắt giữa ký tự UTF-8. */
function tail(text: string, maxBytes: number): string {
  const buf = Buffer.from(text, "utf-8")
  if (buf.byteLength <= maxBytes) return text
  return new TextDecoder("utf-8", { fatal: false }).decode(buf.subarray(buf.byteLength - maxBytes))
}

/**
 * Chạy một gate và ghi bằng chứng ra đĩa.
 * argv LUÔN là mảng — không bao giờ ghép chuỗi shell.
 */
export async function runGate(opts: RunGateOptions): Promise<VerdictFile> {
  const { root, taskId, name, spec, runner } = opts
  const [cmd, ...args] = spec.argv
  const cwd = root
  const startedAt = new Date().toISOString()
  const t0 = Date.now()

  let output = ""
  let exitCode: number | null = null
  let state: GateState
  let reason: string | undefined

  try {
    const res = await execa(cmd!, args, {
      cwd,
      timeout: spec.timeoutMs ?? DEFAULT_TIMEOUT_MS,
      all: true,
      reject: false,
      stripFinalNewline: false,
    })
    output = res.all ?? ""
    exitCode = res.exitCode ?? null

    if (res.timedOut) {
      state = "fail"
      reason = `Gate "${name}" chạy quá thời gian cho phép (${spec.timeoutMs ?? DEFAULT_TIMEOUT_MS}ms).`
    } else if (exitCode === 0) {
      state = "pass"
    } else {
      state = "fail"
    }
  } catch (err) {
    // execa ném khi không spawn được (ENOENT) — đây là "skipped", không phải "fail"
    const e = err as NodeJS.ErrnoException & { shortMessage?: string }
    state = "skipped"
    reason = `Gate "${name}" không chạy được: ${e.shortMessage ?? e.message}. `
      + `Cài công cụ rồi chạy lại, hoặc đặt required: false trong .junto/config.json.`
    output = reason
  }

  const dir = join(taskDir(root, taskId), "verdicts")
  mkdirSync(dir, { recursive: true })
  writeFileSync(join(dir, `${name}.log`), output, "utf-8")

  const verdict: VerdictFile = {
    gate: name,
    argv: spec.argv,
    cwd,
    exitCode,
    state,
    startedAt,
    durationMs: Date.now() - t0,
    outputTail: tail(output, OUTPUT_TAIL_BYTES),
    outputBytes: Buffer.byteLength(output, "utf-8"),
    outputFile: `verdicts/${name}.log`,
    runner,
    ...(reason ? { reason } : {}),
  }

  writeFileSync(join(dir, `${name}.json`), `${JSON.stringify(verdict, null, 2)}\n`, "utf-8")
  return verdict
}
```

```ts
import { existsSync, mkdirSync, renameSync, rmSync, writeFileSync } from "node:fs"
import { join } from "node:path"
import { execa } from "execa"
import {
  juntoDir, readActiveId, readConfig, readTask, setActiveId, taskDir, writeTask,
  type GateStatus, type Size, type Task,
} from "@junto/core"
import type { ToolContext } from "../context.js"

export type TaskToolInput =
  | { action: "start", title: string, size: Size, gates?: string[] }
  | { action: "switch", id: string }
  | { action: "finish" }

const MAX_SLUG = 40

export function newTaskId(title: string, now: Date): string {
  const date = now.toISOString().slice(0, 10)
  const slug = title
    .normalize("NFD")
    .replace(/[\u0300-\u036f]/g, "")
    .replace(/đ/g, "d").replace(/Đ/g, "D")
    .toLowerCase()
    .replace(/[^a-z0-9]+/g, "-")
    .replace(/^-+|-+$/g, "")
    .slice(0, MAX_SLUG)
    .replace(/-+$/, "")
  return `${date}-${slug}`
}

async function currentCommit(root: string): Promise<string | null> {
  try {
    const res = await execa("git", ["rev-parse", "HEAD"], { cwd: root, reject: false })
    return res.exitCode === 0 ? res.stdout.trim() : null
  } catch {
    return null
  }
}

async function start(ctx: ToolContext, input: Extract<TaskToolInput, { action: "start" }>): Promise<string> {
  if (readActiveId(ctx.root) !== null) {
    throw new Error(
      "Đang có task active. junto chỉ giữ một task active mỗi lúc — "
      + "chạy /junto:finish cho task hiện tại trước đã.",
    )
  }

  const config = readConfig(ctx.root)
  const now = new Date()
  const id = newTaskId(input.title, now)
  const iso = now.toISOString()

  const gates: Record<string, GateStatus> = {}
  const names = input.gates ?? Object.keys(config.gates)
  for (const name of names) {
    const spec = config.gates[name]
    if (!spec) continue
    gates[name] = { required: spec.required, verdict: null, stale: false, failStreak: 0 }
  }

  const firstPhase = input.size === "small" ? "build" : "brief"
  const task: Task = {
    schemaVersion: 1,
    id,
    title: input.title,
    size: input.size,
    phase: firstPhase,
    baseCommit: await currentCommit(ctx.root),
    createdAt: iso,
    updatedAt: iso,
    phases: { [firstPhase]: { status: "active", at: iso } },
    gates,
    decisions: [],
    consults: [],
  }

  const dir = taskDir(ctx.root, id)
  mkdirSync(join(dir, "verdicts"), { recursive: true })
  if (!existsSync(join(dir, "brief.md"))) writeFileSync(join(dir, "brief.md"), "", "utf-8")
  writeFileSync(join(dir, "context.jsonl"), "", "utf-8")
  writeTask(ctx.root, task)
  setActiveId(ctx.root, id)

  // .junto/.gitignore — do junto tạo, KHÔNG đụng .gitignore gốc (R1)
  const ignore = join(juntoDir(ctx.root), ".gitignore")
  if (!existsSync(ignore)) writeFileSync(ignore, "*.log\n", "utf-8")

  return `Đã tạo task "${id}" (cỡ ${input.size}), phase ${firstPhase}. `
    + `Gate: ${Object.keys(gates).join(", ") || "chưa có"}.`
}

function finish(ctx: ToolContext): string {
  const id = readActiveId(ctx.root)
  if (id === null) throw new Error("Không có task active để kết thúc.")

  const task = readTask(ctx.root, id)
  const from = taskDir(ctx.root, id)
  const to = join(juntoDir(ctx.root), "archive", id)
  mkdirSync(join(juntoDir(ctx.root), "archive"), { recursive: true })

  const gateLines = Object.entries(task.gates)
    .map(([n, g]) => `- ${n}: ${g.verdict === null ? "chưa chạy" : g.stale ? "hết hạn" : "đã chạy"}${g.required ? " (bắt buộc)" : ""}`)
    .join("\n")
  const summary = `# ${task.title}\n\n`
    + `- id: ${task.id}\n- cỡ: ${task.size}\n- phase cuối: ${task.phase}\n`
    + `- tạo: ${task.createdAt}\n- kết thúc: ${new Date().toISOString()}\n\n`
    + `## Gate\n\n${gateLines || "(không có)"}\n\n`
    + `## Quyết định\n\n${task.decisions.map(d => `- ${d.what} — ${d.why}`).join("\n") || "(không có)"}\n`

  writeFileSync(join(from, "summary.md"), summary, "utf-8")
  rmSync(to, { recursive: true, force: true })
  renameSync(from, to)
  setActiveId(ctx.root, null)

  return `Đã lưu trữ task "${id}" vào .junto/archive/${id}/.`
}

function switchTo(ctx: ToolContext, id: string): string {
  if (!existsSync(join(taskDir(ctx.root, id), "task.json"))) {
    throw new Error(`Không tìm thấy task "${id}" trong .junto/tasks/.`)
  }
  setActiveId(ctx.root, id)
  const task = readTask(ctx.root, id)
  return `Đã chuyển sang task "${id}" (phase ${task.phase}).`
}

export async function taskTool(ctx: ToolContext, input: TaskToolInput): Promise<string> {
  switch (input.action) {
    case "start": return start(ctx, input)
    case "finish": return finish(ctx)
    case "switch": return switchTo(ctx, input.id)
  }
}
```

```ts
import { readActiveId, readConfig, readTask, runGate, writeTask } from "@junto/core"
import type { VerdictFile } from "@junto/core"
import type { ToolContext } from "../context.js"

const FAIL_STREAK_HINT_AT = 3

function render(v: VerdictFile, streak: number): string {
  const head = `## ${v.gate} — ${v.state} (${v.durationMs}ms)`

  if (v.state === "pass") {
    // Bất đối xứng có chủ ý: pass không cần output.
    return `${head}\nĐạt. Output đầy đủ ở ${v.outputFile}.`
  }

  if (v.state === "skipped") {
    return `${head}\n${v.reason ?? "Không chạy được."}\n`
      + `Lưu ý: "skipped" KHÔNG thoả mãn một gate bắt buộc — nó không tính là đạt.`
  }

  const hint = streak >= FAIL_STREAK_HINT_AT
    ? `\n\n⚠ Gate "${v.gate}" đã fail ${streak} lần liên tiếp. `
      + `Cân nhắc khả năng plan sai, không phải code sai.`
    : ""

  return `${head}\nexit ${v.exitCode}. ${v.outputBytes} byte output, đầy đủ ở ${v.outputFile}.\n\n`
    + `\`\`\`\n${v.outputTail}\n\`\`\`${hint}`
}

export async function verifyTool(ctx: ToolContext, input: { gates?: string[] }): Promise<string> {
  const id = readActiveId(ctx.root)
  if (id === null) throw new Error("Không có task active. Chạy /junto:start trước.")

  const config = readConfig(ctx.root)
  const task = readTask(ctx.root, id)
  const names = input.gates ?? Object.keys(task.gates)

  const sections: string[] = []
  for (const name of names) {
    const spec = config.gates[name]
    const status = task.gates[name]
    if (!spec || !status) {
      sections.push(`## ${name} — không có trong .junto/config.json, bỏ qua.`)
      continue
    }

    const verdict = await runGate({ root: ctx.root, taskId: id, name, spec, runner: ctx.runner })

    status.verdict = verdict.outputFile.replace(/\.log$/, ".json")
    status.stale = false
    status.failStreak = verdict.state === "pass" ? 0 : status.failStreak + 1
    sections.push(render(verdict, status.failStreak))
  }

  writeTask(ctx.root, task)
  return sections.join("\n\n")
}
```

```ts
import { existsSync, readFileSync } from "node:fs"
import { join } from "node:path"
import {
  canEnter, readActiveId, readConfig, readTask, taskDir, writeTask,
  type Config, type GateState, type Phase, type Task, type TransitionContext,
} from "@junto/core"
import type { ToolContext } from "../context.js"

/** Đọc các dữ kiện đĩa mà canEnter cần, để bản thân canEnter vẫn thuần. */
export function buildTransitionContext(root: string, task: Task, config: Config): TransitionContext {
  const dir = taskDir(root, task.id)

  const readState = (rel: string | null): GateState | null => {
    if (rel === null) return null
    const p = join(dir, rel)
    if (!existsSync(p)) return null
    try {
      return JSON.parse(readFileSync(p, "utf-8")).state as GateState
    } catch {
      return null
    }
  }

  const verdictStates: Record<string, GateState | null> = {}
  for (const [name, status] of Object.entries(task.gates)) {
    verdictStates[name] = readState(status.verdict)
  }

  const brief = join(dir, "brief.md")
  return {
    briefNonEmpty: existsSync(brief) && readFileSync(brief, "utf-8").trim() !== "",
    planExists: existsSync(join(dir, "plan.md")),
    autoApprove: config.autoApprove,
    verdictStates,
  }
}

export async function advanceTool(ctx: ToolContext, input: { to: Phase }): Promise<string> {
  const id = readActiveId(ctx.root)
  if (id === null) throw new Error("Không có task active. Chạy /junto:start trước.")

  const task = readTask(ctx.root, id)
  const config = readConfig(ctx.root)
  const check = canEnter(task, input.to, buildTransitionContext(ctx.root, task, config))

  if (!check.ok) throw new Error(`Không chuyển sang "${input.to}" được. ${check.reason}`)

  const now = new Date().toISOString()
  const prev = task.phases[task.phase]
  if (prev) task.phases[task.phase] = { ...prev, status: "done", at: now }
  task.phases[input.to] = { ...(task.phases[input.to] ?? {}), status: "active", at: now }
  task.phase = input.to
  writeTask(ctx.root, task)

  return `Task "${id}" chuyển sang phase ${input.to}.`
}
```

```ts
/** Đọc payload JSON mà Claude Code đẩy vào stdin của hook. */
export async function readStdin(): Promise<string> {
  const chunks: Buffer[] = []
  for await (const chunk of process.stdin) chunks.push(Buffer.from(chunk))
  return Buffer.concat(chunks).toString("utf-8")
}

export function parseHookInput(raw: string): Record<string, unknown> {
  try {
    const v = JSON.parse(raw)
    return typeof v === "object" && v !== null ? v as Record<string, unknown> : {}
  } catch {
    return {}
  }
}

/**
 * Hook KHÔNG BAO GIỜ được làm hỏng phiên làm việc.
 * Bọc mọi thứ: lỗi thì im lặng thoát 0.
 */
export async function runHook(fn: (input: Record<string, unknown>) => string | Promise<string>): Promise<void> {
  try {
    const out = await fn(parseHookInput(await readStdin()))
```

```ts
/** Khớp khi prompt BẮT ĐẦU bằng lệnh — nhắc tới nó giữa câu thì không tính. */

export function renderStateBlock(task: Task): string {
  const gates = Object.entries(task.gates).map(([name, g]) => {
    if (g.verdict === null) return `${name}(chưa)`
    if (g.stale) return `${name}(stale)`
    return `${name}(ok)`
  }).join(" ")

  const plan = task.phases.plan?.approvedBy === "user" ? "Plan đã duyệt." : ""
  const step = task.phases[task.phase]?.step
  const of = task.phases[task.phase]?.of
  const progress = step && of ? ` Bước ${step}/${of}.` : ""

  return `<junto task="${task.id}" phase="${task.phase}" size="${task.size}">\n`
    + `${plan}${progress}\n`
    + `Gates: ${gates || "chưa cấu hình"}\n`
    + `</junto>`
}

export function handleState(input: { prompt?: string, cwd?: string }): string {
  const cwd = input.cwd ?? process.cwd()
  const root = findProjectRoot(cwd)
  if (root === null) return ""

  const id = readActiveId(root)
  if (id === null) return ""

  let task: Task
  try {
    task = readTask(root, id)
  } catch {
    return ""
  }

  const prompt = input.prompt ?? ""

  if (isApproval(prompt)) {
    if (task.phase !== "plan") {
      return `<junto>Task "${id}" không ở phase plan (đang ở "${task.phase}"), không có gì để duyệt.</junto>`
    }
    task.phases.plan = { ...(task.phases.plan ?? { status: "active" }), approvedBy: "user" }
    writeTask(root, task)
    return `<junto>Người dùng đã duyệt plan của task "${id}". `
      + `Gọi junto__advance với to="build" để sang phase tiếp theo.</junto>`
  }

  return renderStateBlock(task)
```

```ts
import { relative } from "node:path"
import {
  findProjectRoot, PROTECTED_GLOBS, readActiveId, readConfig, readTask,
  shouldStale, writeTask,
} from "@junto/core"
import { runHook } from "./lib/io.js"

interface HookInput {
  hook_event_name?: string
  cwd?: string
  tool_name?: string
  tool_input?: { file_path?: string, [k: string]: unknown }
}

function toRegExp(glob: string): RegExp {
  const body = glob
    .split("/")
    .map(seg => seg === "**" ? "(?:.*)" : seg.replace(/[.+^${}()|[\]\\]/g, "\\$&").replace(/\*/g, "[^/]*"))
    .join("/")
  return new RegExp(`^${body}$`)
}

export function isProtected(relPath: string): boolean {
  const p = relPath.replace(/\\/g, "/")
  return PROTECTED_GLOBS.some(g => toRegExp(g).test(p))
}

function deny(reason: string): string {
  return JSON.stringify({
    hookSpecificOutput: {
      hookEventName: "PreToolUse",
      permissionDecision: "deny",
      permissionDecisionReason: reason,
    },
  })
}

export function handleGuard(input: HookInput): string {
  const filePath = input.tool_input?.file_path
  if (typeof filePath !== "string" || filePath === "") return ""

  const cwd = input.cwd ?? process.cwd()
  const root = findProjectRoot(cwd)
  if (root === null) return ""

  const rel = relative(root, filePath).replace(/\\/g, "/")
  if (rel.startsWith("..")) return ""

  if (input.hook_event_name === "PreToolUse") {
    if (!isProtected(rel)) return ""
    return deny(
      `junto giữ ${rel} là bằng chứng — chỉ MCP tool ghi được. `
      + `Dùng junto__verify để chạy gate, junto__advance để chuyển phase, junto__task cho vòng đời task.`,
    )
  }

  if (input.hook_event_name === "PostToolUse") {
    const id = readActiveId(root)
    if (id === null) return ""
    try {
      const config = readConfig(root)
      if (!shouldStale(rel, config.staleIgnore)) return ""
      const task = readTask(root, id)
      let changed = false
      for (const g of Object.values(task.gates)) {
        if (g.verdict !== null && !g.stale) { g.stale = true; changed = true }
      }
      if (changed) writeTask(root, task)
    } catch {
      return ""
    }
    return ""
  }

  return ""
}

if (process.argv[1]?.endsWith("guard.js")) {
  await runHook(i => handleGuard(i as HookInput))
}
```

```ts
import { existsSync, readFileSync } from "node:fs"
import { join } from "node:path"
import { findProjectRoot, readActiveId, readTask, taskDir } from "@junto/core"
import { runHook } from "./lib/io.js"

const MAX_DOC_CHARS = 4000

function readDoc(dir: string, name: string): string {
  const p = join(dir, name)
  if (!existsSync(p)) return ""
  return readFileSync(p, "utf-8").trim().slice(0, MAX_DOC_CHARS)
}

export function handleSession(input: { source?: string, cwd?: string }): string {
  const root = findProjectRoot(input.cwd ?? process.cwd())
  if (root === null) return ""

  const id = readActiveId(root)
  if (id === null) return ""

  try {
    const task = readTask(root, id)
    const dir = taskDir(root, id)
    const brief = readDoc(dir, "brief.md")
    const plan = readDoc(dir, "plan.md")
    const decisions = task.decisions.map(d => `- ${d.what} — ${d.why}`).join("\n")

    const parts = [
      `<junto-session task="${task.id}" phase="${task.phase}" size="${task.size}">`,
      `# ${task.title}`,
      brief ? `## Yêu cầu\n${brief}` : "",
      plan ? `## Plan\n${plan}` : "",
      decisions ? `## Quyết định đã chốt\n${decisions}` : "",
      `</junto-session>`,
    ].filter(Boolean)

    return parts.join("\n\n")
  } catch {
    return ""
  }
}

if (process.argv[1]?.endsWith("session.js")) {
  await runHook(i => handleSession(i as { source?: string, cwd?: string }))
```

```ts
import { existsSync, readFileSync } from "node:fs"
import { join } from "node:path"
import { findProjectRoot, readActiveId, taskDir } from "@junto/core"
import { runHook } from "./lib/io.js"

interface Row { file?: string, reason?: string, _example?: unknown }

export function handleHandoff(input: {
  cwd?: string
  tool_name?: string
  tool_input?: Record<string, unknown>
}): string {
  if (input.tool_name !== "Task") return ""

  const root = findProjectRoot(input.cwd ?? process.cwd())
  if (root === null) return ""

  const id = readActiveId(root)
  if (id === null) return ""

  const path = join(taskDir(root, id), "context.jsonl")
  if (!existsSync(path)) return ""

  const rows: Row[] = readFileSync(path, "utf-8")
    .split("\n")
    .map(l => l.trim())
    .filter(l => l !== "")
    .map((l) => { try { return JSON.parse(l) as Row } catch { return {} } })
    .filter(r => r._example === undefined && typeof r.file === "string")

  if (rows.length === 0) return ""

  const lines = rows.map(r => `- ${r.file}${r.reason ? ` — ${r.reason}` : ""}`).join("\n")

  return JSON.stringify({
    hookSpecificOutput: {
      hookEventName: "PreToolUse",
      additionalContext:
        `<junto-spec task="${id}">\nFile liên quan tới task đang chạy:\n${lines}\n</junto-spec>`,
    },
  })
}

if (process.argv[1]?.endsWith("handoff.js")) {
  await runHook(i => handleHandoff(i as Parameters<typeof handleHandoff>[0]))
}
```

```ts
import { mkdirSync, mkdtempSync, readdirSync, statSync, writeFileSync } from "node:fs"
import { tmpdir } from "node:os"
import { join, relative } from "node:path"
import { describe, expect, it } from "vitest"
import { readActiveId } from "@junto/core"
import { taskTool } from "../packages/mcp/src/tools/task.js"
import { verifyTool } from "../packages/mcp/src/tools/verify.js"
import { advanceTool } from "../packages/mcp/src/tools/advance.js"

/** Chụp mọi đường dẫn kèm mtime, đệ quy. */
function snapshot(dir: string): Map<string, number> {
  const out = new Map<string, number>()
  const walk = (d: string) => {
    for (const e of readdirSync(d, { withFileTypes: true })) {
      const p = join(d, e.name)
      if (e.isDirectory()) { out.set(p, 0); walk(p) }
      else out.set(p, statSync(p).mtimeMs)
    }
  }
  walk(dir)
  return out
}

describe("R1 — junto chỉ ghi trong .junto/", () => {
  it("một vòng đời task trọn vẹn không đụng gì ngoài .junto/", async () => {
    const root = mkdtempSync(join(tmpdir(), "junto-boundary-"))
    mkdirSync(join(root, ".junto"), { recursive: true })
    mkdirSync(join(root, "src"), { recursive: true })
    writeFileSync(join(root, "src", "app.ts"), "export const a = 1\n", "utf-8")
    writeFileSync(join(root, "README.md"), "# thử\n", "utf-8")
    writeFileSync(join(root, ".gitignore"), "node_modules/\n", "utf-8")
    writeFileSync(join(root, ".junto", "config.json"), JSON.stringify({
      schemaVersion: 1,
      gates: { tests: { argv: ["node", "-e", "process.exit(0)"], required: true } },
    }), "utf-8")

    const before = snapshot(root)
    const ctx = { root, runner: "boundary@0" }

    await taskTool(ctx, { action: "start", title: "Thử ranh giới", size: "small" })
    await advanceTool(ctx, { to: "verify" })
    await verifyTool(ctx, {})
    await advanceTool(ctx, { to: "done" })
    await taskTool(ctx, { action: "finish" })

    const after = snapshot(root)
    const junto = join(root, ".junto")

    const touched: string[] = []
    for (const [path, mtime] of after) {
      if (path.startsWith(junto)) continue
      const prev = before.get(path)
      if (prev === undefined || prev !== mtime) touched.push(relative(root, path))
    }
    for (const path of before.keys()) {
      if (path.startsWith(junto)) continue
      if (!after.has(path)) touched.push(`ĐÃ XOÁ: ${relative(root, path)}`)
    }

    expect(touched, `junto đã chạm file ngoài .junto/: ${touched.join(", ")}`).toEqual([])
  })

  it("không tạo .junto ở HOME", async () => {
    const home = process.env.HOME ?? process.env.USERPROFILE
    expect(home).toBeTruthy()
    const root = mkdtempSync(join(tmpdir(), "junto-home-"))
    mkdirSync(join(root, ".junto"), { recursive: true })
    writeFileSync(join(root, ".junto", "config.json"), JSON.stringify({ schemaVersion: 1, gates: {} }), "utf-8")

    const homeBefore = snapshot(join(home!, ".claude")).size
    await taskTool({ root, runner: "t@0" }, { action: "start", title: "X", size: "small" })
    expect(readActiveId(root)).not.toBeNull()
    expect(snapshot(join(home!, ".claude")).size).toBe(homeBefore)
  })
})
```
